    <script>
        // 1. Cơ chế tải ảnh thực tế từ Google Drive thông minh (Tự động chuyển đổi link nếu trình duyệt chặn cookie)
        function handleDriveImgError(imgElement, fallbackPlaceholderUrl) {
            const driveId = imgElement.getAttribute('data-drive-id');
            const currentAttempt = parseInt(imgElement.getAttribute('data-attempt') || '0', 10);

            if (driveId && currentAttempt === 0) {
                imgElement.setAttribute('data-attempt', '1');
                imgElement.src = 'https://lh3.googleusercontent.com/d/' + driveId + '=w1000';
            } else if (driveId && currentAttempt === 1) {
                imgElement.setAttribute('data-attempt', '2');
                imgElement.src = 'https://drive.google.com/uc?export=view&id=' + driveId;
            } else {
                imgElement.onerror = null;
                imgElement.src = fallbackPlaceholderUrl;
            }
        }

        // 2. Điều khiển 2 ảnh Banner tự trượt chuyên nghiệp
        let currentSlideIndex = 0;
        const totalSlides = 2;
        let slideInterval = null;

        function showBanner(index) {
            currentSlideIndex = (index + totalSlides) % totalSlides;
            for (let i = 0; i < totalSlides; i++) {
                const slideEl = document.getElementById('slide-' + i);
                const dotEl = document.getElementById('dot-' + i);
                if (i === currentSlideIndex) {
                    slideEl.classList.add('active');
                    dotEl.className = 'w-6 h-2 rounded-full bg-brand-amber transition-all';
                } else {
                    slideEl.classList.remove('active');
                    dotEl.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
                }
            }
        }

        function nextBanner() {
            showBanner(currentSlideIndex + 1);
            resetBannerTimer();
        }

        function prevBanner() {
            showBanner(currentSlideIndex - 1);
            resetBannerTimer();
        }

        function goToBanner(index) {
            showBanner(index);
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (slideInterval) clearInterval(slideInterval);
            slideInterval = setInterval(() => {
                showBanner(currentSlideIndex + 1);
            }, 5000);
        }

        // 3. Đóng/Mở Menu trên điện thoại
        function toggleMobileMenu() {
            const panel = document.getElementById('mobileMenuPanel');
            panel.classList.toggle('hidden');
        }

        // 4. Lọc sản phẩm theo Danh mục
        function filterCategory(categoryKey) {
            const cards = document.querySelectorAll('.product-card');
            cards.forEach(card => {
                const cardCat = card.getAttribute('data-category');
                if (categoryKey === 'all' || cardCat === categoryKey) {
                    card.classList.remove('hidden');
                } else {
                    card.classList.add('hidden');
                }
            });

            // Cập nhật trạng thái nút danh mục
            const buttons = document.querySelectorAll('.cat-filter-btn');
            buttons.forEach(btn => {
                btn.classList.remove('bg-brand-amber', 'text-white', 'border-brand-amber', 'font-bold');
                btn.classList.add('bg-slate-50', 'text-brand-navy', 'border-slate-200', 'font-semibold');
            });

            const activeBtn = document.getElementById('cat-btn-' + categoryKey);
            if (activeBtn) {
                activeBtn.classList.remove('bg-slate-50', 'text-brand-navy', 'border-slate-200', 'font-semibold');
                activeBtn.classList.add('bg-brand-amber', 'text-white', 'border-brand-amber', 'font-bold');
            }
        }

        // 5. Tìm kiếm sản phẩm nhanh
        function filterProductsByKeyword(keyword) {
            const normalized = keyword.trim().toLowerCase();
            const cards = document.querySelectorAll('.product-card');
            cards.forEach(card => {
                const text = card.innerText.toLowerCase();
                if (!normalized || text.includes(normalized)) {
                    card.classList.remove('hidden');
                } else {
                    card.classList.add('hidden');
                }
            });
            document.getElementById('cat-xay-dung').scrollIntoView({ behavior: 'smooth' });
        }

        function filterProductsBySearch() {
            const val = document.getElementById('searchInput').value;
            filterProductsByKeyword(val);
        }

        function filterProductsByMobileSearch() {
            const val = document.getElementById('mobileSearchInput').value;
            filterProductsByKeyword(val);

        // 6. Chọn sản phẩm điền sẵn vào Form Yêu Cầu Báo Giá
        function selectProductForQuote(productName) {
            const selectEl = document.getElementById('custProduct');
            if (selectEl) {
                for (let i = 0; i < selectEl.options.length; i++) {
                    if (selectEl.options[i].value === productName) {
                        selectEl.selectedIndex = i;
                        break;
                    }
                }
            }
            const formSection = document.getElementById('form-bao-gia');
            if (formSection) {
                formSection.scrollIntoView({ behavior: 'smooth' });
                setTimeout(() => {
                    document.getElementById('custName').focus();
                }, 500);
            }
        }
